33template <typename Container, typename ObjType>
34TI binary_search(Container& container, TI first, TI last, ObjType &object)
35{
36 if( first >= last )
37 return first;
38 while( first < last )
39 {
40 TI mid = (first+last)/2;
41 if( object == (ObjType)container[mid ] )
42 return mid;
43 if( object > (ObjType)container[mid ] )
44 first = mid+1;
45 else
46 last = mid;
47 }
48 if( object <= (ObjType)container[first] )
49 return first;
50 return last;
51}

241template <typename Traits>
242bt_ErrorCode CBTreePage<Traits>::Insert(const keyType& key, const ObjIDType ObjID)
243{
244 scoped_lock<mutex> lock(m_mtx);
245 TI pos = binary_search(m_Keys, 0, m_KeyCount, key);
246 bt_ErrorCode error = bt_ok;
247
248 if( pos < m_KeyCount && (keyType)m_Keys[pos] == key && m_Unique)
249 return bt_duplicate; // this key is duplicate
250
251 if( !m_SubPages[pos] ) // this is a leave
252 {
253 ::insert_at(m_Keys, Node(key, ObjID), pos);
254 NumberOfKeys()++;
255 if( Overflow() )
256 return bt_overflow;
257 return bt_ok;
258 }
259 // recursive insertion
260 error = m_SubPages[pos]->Insert(key, ObjID);
261 if( error == bt_overflow )
262 {
263 if( !Redistribute1(pos) )
264 SplitChild(pos);
265 if( Overflow() ) // Propagate overflow
266 return bt_overflow;
267 return bt_ok;
268 }
269 return bt_ok;
270}

272template <typename Traits>
273TB CBTreePage<Traits>::Redistribute1(TI &pos)
274{
275 if( m_SubPages[pos]->Underflow() )
276 { // nkol = Number of keys on left brother, nkor = Number of keys on right brother
277 TI nkol = 0,
278 nkor = 0;
279 // is this the first element or there are more elements on right brother
280 if( pos > 0 )
281 nkol = m_SubPages[pos-1]->NumberOfKeys();
282 if( pos < NumberOfKeys() )
283 nkor = m_SubPages[pos+1]->NumberOfKeys();
284
285 if( nkol > nkor ){
286 if( m_SubPages[pos-1]->NumberOfKeys() > m_SubPages[pos-1]->MinNumberOfKeys() )
287 RedistributeL2R(pos-1); // bring elements from left brother
288 else{
289 if( pos == NumberOfKeys() )
290 --pos;
291 return false;
292 }
293 }
294 else //nkol < nkor )
295 if( m_SubPages[pos+1]->NumberOfKeys() > m_SubPages[pos+1]->MinNumberOfKeys() )
296 RedistributeR2L(pos+1); // bring elements from right brother
297 else{
298 if( pos == 0 )
299 ++pos;
300 return false;
301 }
302 }
303 else // it is due to overflow
304 {
305 TI fcol = GetFreeCellsOnLeft(pos), // Free Cells On Left
306 fcor = GetFreeCellsOnRight(pos); // Free Cells On Right
307
308 if( !fcol && !fcor && m_SubPages[pos]->IsFull() )
309 return false;
310 if( fcol > fcor ) // There is more space on left
311 RedistributeR2L(pos);
312 else
313 RedistributeL2R(pos);
314
315 }
316 return true;
317}

322template <typename Traits>
323TB CBTreePage<Traits>::Redistribute2(TI pos)
324{
325 assert( pos > 0 && pos < NumberOfKeys() );
326 assert( m_SubPages[pos-1] != 0 && m_SubPages[pos] != 0 && m_SubPages[pos+1] != 0 );
327 assert( m_SubPages[pos-1]->Underflow() ||
328 m_SubPages[ pos ]->Underflow() ||
329 m_SubPages[pos+1]->Underflow() );
330
331 if( m_SubPages[pos-1]->Underflow() )
332 { // Rotate R2L
333 RedistributeR2L(pos+1);
334 RedistributeR2L(pos);
335 if( m_SubPages[pos-1]->Underflow() )
336 return false;
337 }
338 else if( m_SubPages[pos+1]->Underflow() )
339 { // Rotate L2R
340 RedistributeL2R(pos-1);
341 RedistributeL2R(pos);
342 if( m_SubPages[pos+1]->Underflow() )
343 return false;
344 }
345 else // The problem is exactly at pos !
346 {
347 // Rotate L2R
348 RedistributeL2R(pos-1);
349 RedistributeR2L(pos+1);
350 if( m_SubPages[pos]->Underflow() )
351 return false;
352 }
353 return true;
354}

356template <typename Traits>
357void CBTreePage<Traits>::RedistributeR2L(TI pos)
358{
359 BTPage *pSource = m_SubPages[ pos ],
360 *pTarget = m_SubPages[pos-1];
361
362 while(pSource->GetNumberOfKeys() > pSource->MinNumberOfKeys() &&
363 pTarget->GetNumberOfKeys() < pSource->GetNumberOfKeys() )
364 {
365 // Move from this page to the down-left page \/
366 ::insert_at(pTarget->m_Keys, m_Keys[pos-1], pTarget->NumberOfKeys()++);
367 // Move the pointer leftest pointer to the rightest position
368 ::insert_at(pTarget->m_SubPages, pSource->m_SubPages[0], pTarget->NumberOfKeys());
369
370 // Move the leftest element to the root
371 m_Keys[pos-1] = pSource->m_Keys[0];
372
373 // Remove the leftest element from rigth page
374 ::remove(pSource->m_Keys , 0);
375 ::remove(pSource->m_SubPages, 0);
376 pSource->NumberOfKeys()--;
377 }

403template <typename Traits>
404void CBTreePage<Traits>::SplitChild(TI pos)
405{
406 // FIRST: deciding the second page to split
407 BTPage *pChild1 = 0, *pChild2 = 0;
408 if( pos > 0 ) // is left page full ?
409 if( m_SubPages[pos-1]->IsFull() )
410 {
411 pChild1 = m_SubPages[pos-1];
412 pChild2 = m_SubPages[pos--];
413 }
414 if( pos < GetNumberOfKeys() ) // is right page full ?
415 if( m_SubPages[pos+1]->IsFull() )
416 {
417 pChild1 = m_SubPages[pos];
418 pChild2 = m_SubPages[pos+1];
419 }
420
421 //TI nKeys = pChild1->GetNumberOfKeys() + pChild2->GetNumberOfKeys() + 1;
422
423 // SECOND: copy both pages to a temporal one
424 // Create two tmp vector
425 vector<Node> tmpKeys;
426 //tmpKeys.resize(nKeys);
427 vector<BTPage *> tmpSubPages;
428 //tmpKeys.resize(nKeys+1);
429
430 // Prepara el vectpor unificado de las 2 paginas a ser divididas en 3
431 // copy from left child
432 MovePage(pChild1, tmpKeys, tmpSubPages);
433 // copy a key from parent
434 tmpKeys .push_back(m_Keys[pos]);
435
436 // copy from right child
437 MovePage(pChild2, tmpKeys, tmpSubPages);
438
439 BTPage *pChild3 = 0;
440 Node oi1, oi2;
441 SplitPageInto3(tmpKeys, tmpSubPages, pChild1, pChild2, pChild3, oi1, oi2);
442
443 // copy the first element to the root
444 m_Keys [pos] = oi1;
445 m_SubPages[pos] = pChild1;
446
447 // copy the second element to the root
448 ::insert_at(m_Keys, oi2, pos+1);
449 ::insert_at(m_SubPages, pChild2, pos+1);
450 NumberOfKeys()++;
451
452 m_SubPages[pos+2] = pChild3;
453}

467 pChild1 = new BTPage(m_MaxKeysForChilds, m_Unique);

484
485 if( !pChild2 )

487 pChild2->clear();
488 // copy 1/3 to the second child
489 nKeys += (tmpKeys.size()-2)/3 + 1;

493 pChild2->m_Keys [j] = tmpKeys [i];

496 }
497 pChild2->m_SubPages[j] = tmpSubPages[i];
498

501
502 // copy 1/3 to the third child
503 if( !pChild3 )
504 pChild3 = new BTPage(m_MaxKeysForChilds, m_Unique);

506 nKeys = tmpKeys.size();
507 for(j = 0; i < nKeys; i++, j++)

540{

557 return m_SubPages[pos]->Search(key, ObjID);
558 return false;

650template <typename Traits>
651bt_ErrorCode CBTreePage<Traits>::Remove(const keyType &key, const ObjIDType ObjID)
652{
653 scoped_lock<mutex> lock(m_mtx);
654 bt_ErrorCode error = bt_ok;
655 TI pos = binary_search(m_Keys, 0, m_KeyCount, key);
656 if( pos < NumberOfKeys() && key == m_Keys[pos].key /*&& m_Keys[pos].m_ObjID == ObjID*/) // We found it !
657 {
658 // This is a leave: First
659 if( !m_SubPages[pos+1] ) // This is a leave ? FIRST CASE !
660 {
661 ::remove(m_Keys, pos);
662 NumberOfKeys()--;
663 if( Underflow() )
664 return bt_underflow;
665 return bt_ok;
666 }
667
668 // We FOUND IT BUT it is NOT a leave ? SECOND CASE !
669 {
670 // Get the first element from right branch
671 Node &rFirstFromRight = m_SubPages[pos+1]->GetFirstNode();
672 // change with a leave
673 swap(m_Keys[pos], rFirstFromRight);
674 // Remove it from this leave
675
676 //Print(cout);
677 error = m_SubPages[++pos]->Remove(key, ObjID);
678 }
679 }
680 else if( pos == NumberOfKeys() ) // it is not here, go by the last branch
681 error = m_SubPages[pos]->Remove(key, ObjID);
682 else if( key <= m_Keys[pos].key ){ // = is because identical keys are inserted on left (see Insert)
683 if( m_SubPages[pos] )
684 error = m_SubPages[pos]->Remove(key, ObjID);
685 else
686 return bt_nofound;
687 }
688 if( error == bt_underflow ){
689 // THIRD CASE: After removing the element we have an underflow
690 //Print(cout);
691 if( TreatUnderflow(pos) )
692 return bt_ok;
693 // FOURTH CASE: it was not possible to redistribute -> Merge
694 if( IsRoot() && NumberOfKeys() == 2 )
695 return MergeRoot();
696 return Merge(pos);
697 }
698 if( error == bt_nofound )
699 return bt_nofound;
700 return bt_ok;
701}

704template <typename Traits>
705bt_ErrorCode CBTreePage<Traits>::Merge(TI pos)
706{
707 assert( m_SubPages[pos-1]->NumberOfKeys() +
708 m_SubPages[ pos ]->NumberOfKeys() +
709 m_SubPages[pos+1]->NumberOfKeys() ==
710 3*m_SubPages[ pos ]->MinNumberOfKeys() - 1);
711
712 // FIRST: Put all the elements into a vector
713 vector<Node> tmpKeys;
714 //tmpKeys.resize(nKeys);
715 vector<BTPage *> tmpSubPages;
716
717 BTPage *pChild1 = m_SubPages[pos-1],
718 *pChild2 = m_SubPages[ pos ],
719 *pChild3 = m_SubPages[pos+1];
720 MovePage(pChild1, tmpKeys, tmpSubPages);
721 tmpKeys .push_back(m_Keys[pos-1]);
722 MovePage(pChild2, tmpKeys, tmpSubPages);
723 tmpKeys .push_back(m_Keys[ pos ]);
724 MovePage(pChild3, tmpKeys, tmpSubPages);
725 pChild3->Destroy();;
726
727 // Move 1/2 elements to pChild1
728 TI nKeys = pChild1->GetFreeCells();
729 TI i = 0;
730 for( ; i < nKeys ; i++ )
731 {
732 pChild1->m_Keys [i] = tmpKeys [i];
733 pChild1->m_SubPages[i] = tmpSubPages[i];
734 pChild1->NumberOfKeys()++;
735 }
736 pChild1->m_SubPages[i] = tmpSubPages[i];
737
738 m_Keys [pos-1] = tmpKeys[i];
739 m_SubPages[pos-1] = pChild1;
740
741 ::remove(m_Keys , pos);
742 ::remove(m_SubPages, pos);
743 NumberOfKeys()--;
744
745 nKeys = pChild2->GetFreeCells();
746 TI j = ++i;
747 for(i = 0 ; i < nKeys ; i++, j++ )
748 {
749 pChild2->m_Keys [i] = tmpKeys [j];
750 pChild2->m_SubPages[i] = tmpSubPages[j];
751 pChild2->NumberOfKeys()++;
752 }
753 pChild2->m_SubPages[i] = tmpSubPages[j];
754 m_SubPages[ pos ] = pChild2;
755
756 if( Underflow() )
757 return bt_underflow;
758 return bt_ok;
