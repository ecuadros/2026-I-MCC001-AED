232template <typename Traits>
233TB BTree<Traits>::Insert(const keyType key, const ObjIDType ObjID)
234{
235 scoped_lock<mutex> lock(m_mtx);
236 bt_ErrorCode error = m_Root.Insert(key, ObjID);
237 if( error == bt_duplicate )
238 return false;
239 m_NumKeys++;
240 if( error == bt_overflow )
241 {
242 m_Root.SplitRoot();
243 m_Height++;
244 }
245 return true;
